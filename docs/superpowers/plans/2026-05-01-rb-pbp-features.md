**Goal:** Promote the 4 PBP team-level features greenlit by PR #20's family probe (`pace_l4`, `proe_l4`, `team_ayps_l4`, `team_def_epa_resid_l4`) into production for RB only — append them to `RbFeaturesSchema`, compute + join inside `build_rb_features`, refresh the RB cache, and run the full backtest + adoption gate to verify the probe's ADOPT prediction holds. **Architecture:** Extract `attach_pbp_family_features(index, pbp) → DataFrame` from the existing `build_pbp_family_overrides` assembler so the same compute + 4-merge logic is reused by both the override generator and the production `build_rb_features` builder. RB builder builds a `(gsis_id, season, week, team, opp)` index from `depth_charts × schedules`, calls the helper, merges 4 columns into the existing RB feature frame, validates against the extended schema. Decision is bound to the dual-run gate's verdict on `(BaselineModel, RB)`: ADOPT → ship; MARGINAL / DO_NOT_ADOPT → revert (keep the helper extraction).

**Tech Stack:** Python 3.13, pandas, pyarrow, pandera, pytest (no network), mypy strict, ruff. PBP / weekly_stats / depth_charts / schedules read via `projections.store.read_partition`.

```python
def attach_pbp_family_features(
    index: pd.DataFrame,
    pbp: pd.DataFrame,
) -> pd.DataFrame:
    """Attach the 4 PBP family features to a player-team-week index.

    Args:
        index: ``(gsis_id, season, week, team, opp)`` — one row per
            player-week. Team codes are assumed canonical per the ingest
            schemas (``DepthChartsSchema`` / ``SchedulesSchema`` both validate
            against ``_TEAM_VALUES``).
        pbp: PBP frame matching ``PbpSchema``, projected to or wider than
            ``_PBP_COLUMNS_USED``. Must include the seasons spanning the
            index plus one prior season for trailing-4 backfill.

    Returns:
        A copy of ``index`` with 4 columns appended in order:
        ``pace_l4``, ``proe_l4``, ``team_ayps_l4``, ``team_def_epa_resid_l4``.
        Row count equals ``len(index)``; all 4 columns are float64 nullable
        (NaN where trailing-4 has fewer than 4 prior games).

    pace / proe / team_ayps join on the player's TEAM; team_def_epa_resid
    joins on the player's OPPONENT.

    Empty ``pbp`` short-circuits to all-NaN columns — same shape as a
    successful call where every row's trailing-4 has fewer than 4 prior
    games. Schema ``nullable=True`` covers this.
    """
    if pbp.empty:
        out = index.copy()
        for col in ("pace_l4", "proe_l4", "team_ayps_l4", "team_def_epa_resid_l4"):
            out[col] = pd.array([pd.NA] * len(out), dtype="Float64").astype(float)
        return out

    pbp_proj = pbp[list(_PBP_COLUMNS_USED)]
    pace = compute_team_pace(pbp_proj)
    proe = compute_team_proe(pbp_proj)
    ayps = compute_team_ayps(pbp_proj)
    def_resid = compute_team_def_epa_residual(pbp_proj)

    out = index.merge(pace, on=["team", "season", "week"], how="left")
    out = out.merge(proe, on=["team", "season", "week"], how="left")
    out = out.merge(ayps, on=["team", "season", "week"], how="left")
    out = out.merge(
        def_resid.rename(columns={"team": "opp"}),
        on=["opp", "season", "week"],
        how="left",
    )
    return out
```

```python
def build_pbp_family_overrides(
    pbp: pd.DataFrame,
    player_team_week_index: pd.DataFrame,
) -> pd.DataFrame:
    """Public assembler. Returns the 4-column override frame ready to write.

    [docstring unchanged — keep the existing one]
    """
    bad_ids = [g for g in player_team_week_index["gsis_id"].dropna() if not _GSIS_RE.match(str(g))]
    if bad_ids:
        raise ValueError(
            f"invalid gsis_id format(s): {bad_ids[:3]} (and {max(0, len(bad_ids) - 3)} more)"
        )

    dup_mask = player_team_week_index.duplicated(subset=["gsis_id", "season", "week"], keep=False)
    if dup_mask.any():
        n_dup = int(dup_mask.sum())
        raise ValueError(f"duplicate (gsis_id, season, week) keys in index: {n_dup} rows")

    out = attach_pbp_family_features(player_team_week_index, pbp)

    if len(out) != len(player_team_week_index):
        raise AssertionError(
            f"row count mismatch: input index had {len(player_team_week_index)} rows, "
            f"output has {len(out)}; suggests a many-to-many merge regression"
        )

    return out[
        [
            "gsis_id",
            "season",
            "week",
            "pace_l4",
            "proe_l4",
            "team_ayps_l4",
            "team_def_epa_resid_l4",
        ]
    ].reset_index(drop=True)
```
